1!> Minimal magnetic-field-only state for static field construction.
2!>
3!> The B-only path deliberately registers only the magnetic components as
4!> conservative/primitive variables. It is intended for static potential
5!> (and other field-construction) cases; it does not provide a time-evolving
6!> induction equation. The legacy MHD-embedded magnetofriction path is
7!> intentionally unaffected by this module.

173 subroutine bfield_physical_units()
174 use mod_global_parameters
175 use mod_eos_container, only: eos
176
177 double precision :: mp,kB,miu0,c_lightspeed
178 double precision :: a,b,he
179 character(len=std_len) :: eos_type
180
181 if(si_unit) then
182 mp=mp_si
183 kb=kb_si
184 miu0=miu0_si
185 const_sigmasb=sigma_sb_si
186 c_lightspeed=c_si
187 else
188 mp=mp_cgs
189 kb=kb_cgs
190 miu0=4.d0*dpi
191 const_sigmasb=sigma_sb_cgs
192 c_lightspeed=const_c
193 end if
194
195 he=0.1d0
196 eos_type='FI'
197 if(allocated(eos)) then
198 he=eos%He_abundance
199 eos_type=eos%eos_type
200 end if
201
202 if(eos_type=='LTE') then
203 a=1.d0
204 b=1.d0
205 else
206 a=1.d0+4.d0*he
207 if(eos_type=='PI') then
208 b=1.d0+bfield_h_ion_fr+he*(bfield_he_ion_fr*&
209 (bfield_he_ion_fr2+1.d0)+1.d0)
210 else
211 b=2.d0+3.d0*he
212 end if
213 end if
214
215 if(unit_density/=1.d0 .or. unit_numberdensity/=1.d0) then
216 if(unit_density/=1.d0) then
217 unit_numberdensity=unit_density/(a*mp)
218 else if(unit_numberdensity/=1.d0) then
219 unit_density=a*mp*unit_numberdensity
220 end if
221 if(unit_temperature/=1.d0) then
222 unit_pressure=b*unit_numberdensity*kb*unit_temperature
223 unit_velocity=sqrt(unit_pressure/unit_density)
224 unit_magneticfield=sqrt(miu0*unit_pressure)
225 if(unit_length/=1.d0) then
226 unit_time=unit_length/unit_velocity
227 else if(unit_time/=1.d0) then
228 unit_length=unit_velocity*unit_time
229 end if
230 else if(unit_magneticfield/=1.d0) then
231 unit_pressure=unit_magneticfield**2/miu0
232 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
233 unit_velocity=sqrt(unit_pressure/unit_density)
234 if(unit_length/=1.d0) then
235 unit_time=unit_length/unit_velocity
236 else if(unit_time/=1.d0) then
237 unit_length=unit_velocity*unit_time
238 end if
239 else if(unit_pressure/=1.d0) then
240 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
241 unit_velocity=sqrt(unit_pressure/unit_density)
242 unit_magneticfield=sqrt(miu0*unit_pressure)
243 if(unit_length/=1.d0) then
244 unit_time=unit_length/unit_velocity
245 else if(unit_time/=1.d0) then
246 unit_length=unit_velocity*unit_time
247 end if
248 else if(unit_velocity/=1.d0) then
249 unit_pressure=unit_density*unit_velocity**2
250 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
251 unit_magneticfield=sqrt(miu0*unit_pressure)
252 if(unit_length/=1.d0) then
253 unit_time=unit_length/unit_velocity
254 else if(unit_time/=1.d0) then
255 unit_length=unit_velocity*unit_time
256 end if
257 else if(unit_time/=1.d0) then
258 unit_velocity=unit_length/unit_time
259 unit_pressure=unit_density*unit_velocity**2
260 unit_magneticfield=sqrt(miu0*unit_pressure)
261 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
262 end if
263 else if(unit_temperature/=1.d0) then
264 if(unit_magneticfield/=1.d0) then
265 unit_pressure=unit_magneticfield**2/miu0
266 unit_numberdensity=unit_pressure/(b*unit_temperature*kb)
267 unit_density=a*mp*unit_numberdensity
268 unit_velocity=sqrt(unit_pressure/unit_density)
269 if(unit_length/=1.d0) then
270 unit_time=unit_length/unit_velocity
271 else if(unit_time/=1.d0) then
272 unit_length=unit_velocity*unit_time
273 end if
274 else if(unit_pressure/=1.d0) then
275 unit_magneticfield=sqrt(miu0*unit_pressure)
276 unit_numberdensity=unit_pressure/(b*unit_temperature*kb)
277 unit_density=a*mp*unit_numberdensity
278 unit_velocity=sqrt(unit_pressure/unit_density)
279 if(unit_length/=1.d0) then
280 unit_time=unit_length/unit_velocity
281 else if(unit_time/=1.d0) then
282 unit_length=unit_velocity*unit_time
283 end if
284 end if
285 else if(unit_magneticfield/=1.d0) then
286 if(unit_velocity/=1.d0) then
287 unit_pressure=unit_magneticfield**2/miu0
288 unit_numberdensity=unit_pressure/(b*unit_temperature*kb)
289 unit_density=a*mp*unit_numberdensity
290 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
291 if(unit_length/=1.d0) then
292 unit_time=unit_length/unit_velocity
293 else if(unit_time/=1.d0) then
294 unit_length=unit_velocity*unit_time

297 unit_pressure=unit_magneticfield**2/miu0
298 unit_velocity=unit_length/unit_time
299 unit_density=unit_pressure/unit_velocity**2
300 unit_numberdensity=unit_density/(a*mp)
301 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
302 end if
303 else if(unit_pressure/=1.d0) then
304 if(unit_velocity/=1.d0) then
305 unit_magneticfield=sqrt(miu0*unit_pressure)
306 unit_density=unit_pressure/unit_velocity**2
307 unit_numberdensity=unit_density/(a*mp)
308 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
309 if(unit_length/=1.d0) then
310 unit_time=unit_length/unit_velocity
311 else if(unit_time/=1.d0) then
312 unit_length=unit_velocity*unit_time
313 end if
314 else if(unit_time/=0.d0) then
315 unit_magneticfield=sqrt(miu0*unit_pressure)
316 unit_velocity=unit_length/unit_time
317 unit_density=unit_pressure/unit_velocity**2
318 unit_numberdensity=unit_density/(a*mp)
319 unit_temperature=unit_pressure/(b*unit_numberdensity*kb)
320 end if
321 end if
322
323 c_norm=c_lightspeed/unit_velocity
324 unit_charge=unit_magneticfield*unit_length**2/unit_velocity/miu0
325 if(.not.si_unit) unit_charge=unit_charge*const_c
326 unit_mass=unit_density*unit_length**3
