135 subroutine small_values_average(ixI^L, ixO^L, w, x, w_flag, windex)
136 use mod_global_parameters
137 integer, intent(in) :: ixi^l, ixo^l
138 logical, intent(in) :: w_flag(ixi^s,1:nw)
139 double precision, intent(inout) :: w(ixi^s, 1:nw)
140 double precision, intent(in) :: x(ixi^s, 1:ndim)
141 integer, optional, intent(in) :: windex
142 integer :: iw, kxo^l, ix^d, i, nwstart, nwend
143
144 type(repair_warning_state), save :: warning_state

149 fallback_events=0_int64
150 if(present(windex)) then
151 nwstart=windex
152 nwend=windex
153 else
154 nwstart=1
155 nwend=nw
156 end if
157
158 do iw = nwstart, nwend
159 {do ix^db= ixo^lim^db\}
160 ! point with local failure identified by w_flag
161 if (w_flag(ix^d,iw)) then
162 ! verify in cube with border width small_values_daverage the presence of
163 ! cells where all went ok
164 do i = 1, max(small_values_daverage, 1)
165 {kxomin^d= max(ix^d-i, iximin^d);
166 kxomax^d= min(ix^d+i, iximax^d);\}
167 ! in case cells are fine within smaller cube than
168 ! the userset small_values_daverage: use that smaller cube
169 if(any(w_flag(kxo^s,iw) .eqv. .false.)) exit
170 end do
171
172 if(any(w_flag(kxo^s,iw) .eqv. .false.)) then
173 ! within surrounding cube, cells without problem were found
174
175 ! faulty cells are corrected by averaging here
176 ! only average those which were ok and replace faulty cells
177 if(small_values_fix_iw(iw)) then
178 w(ix^d, iw) = sum(w(kxo^s, iw), w_flag(kxo^s,iw) .eqv. .false.)&
179 / count(w_flag(kxo^s,iw) .eqv. .false.)
180 end if
181 else
182 fallback_events=fallback_events+1_int64
183 if(fallback_events==1_int64) then
184 first_iw=iw
185 first_x=x(ix^d,1:ndim)
186 first_before=w(ix^d,iw)
187 end if
188 if(iw==iw_e) w(ix^d, iw)=small_pressure
189 if(iw==iw_rho) w(ix^d, iw)=small_density
190 if(fallback_events==1_int64) first_after=w(ix^d,iw)
191 end if
192 end if
193 {enddo^d&\}
194 end do
195 ! Other variables can remain unchanged in this existing fallback path.
196 if(fallback_events>0_int64) call report_repair_warning(warning_state, &
