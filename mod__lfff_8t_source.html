1!> Program to extrapolate linear force-free fields in 3D Cartesian coordinates,
2!> based on exact Green function method (Chiu & Hilton 1977 ApJ 212,873).
3!>
4!> Usage:
5!> 1 In the subroutine usr_set_parameters of mod_usr.t:
6!> To extrapolate a linear force free field from a observed magnetogram
7!> prepared in a data file, e.g., 'hmiM720sxxxx.dat' replace
8!> call init_bc_fff_data('hmiM720sxxxx.dat',unit_length,unit_magneticfield)
9!> 'hmiM720sxxxx.dat' must be a binary file containing nx1,nx2,xc1,xc2,dxm1,
10!> dxm2, Bz0(nx1,nx2). Integers nx1 and nx2 give the resolution of the
11!> uniform-grid magentogram. Others are double-precision floats. xc1 and xc2
12!> are coordinates of the central point of the magnetogram. dxm1 and dxm2
13!> are the cell sizes for each direction, Bz0 is the vertical conponent
14!> of magetic field on the solar surface from observations.
15!>2 In the subroutine usr_init_one_grid of mod_usr.t,
16!> add lines like:
17!>
18!> double precision :: Bf(ixG^S,1:ndir), alpha, zshift
19!>
20!> alpha=0.d0 ! potential field
21!> !alpha=0.08d0 ! non-potential linear force-free field
22!> zshift=0.05d0 ! lift your box zshift heigher to the bottom magnetogram
23!> call calc_lin_fff(ixG^L,ix^L,Bf,x,alpha,zshift)
24!>
25!>3 Notice that the resolution of input magnetogram must be better than the best
26!> resolution of your AMR grid to have a good behavior close to the bottom layer

170 !> Check a constant-alpha magnetogram and optionally remove its core mean.

173 subroutine lfff_balance_bottom_flux(bz,treatment,max_imbalance,&
174 imbalance_before,imbalance_after,mean_correction,status)
175 double precision, intent(inout) :: bz(:,:)
176 character(len=*), intent(in) :: treatment
177 double precision, intent(in) :: max_imbalance
178 double precision, intent(out) :: imbalance_before,imbalance_after
179 double precision, intent(out) :: mean_correction
180 integer, intent(out) :: status
181
182 double precision :: unsigned_flux
183
184 mean_correction=0.d0
185 unsigned_flux=sum(dabs(bz))
186 imbalance_before=dabs(sum(bz))/max(unsigned_flux,tiny(1.d0))
187 imbalance_after=imbalance_before
188 status=0
189
190 if(max_imbalance<0.d0 .or. max_imbalance>1.d0) then
191 status=3
192 return
193 end if
194
195 select case(trim(adjustl(treatment)))
196 case('strict')
197 if(imbalance_before>lfff_flux_balance_tolerance) status=4
198 case('subtract_mean')
199 if(imbalance_before>max_imbalance) then
200 status=5
201 return
202 end if
203 if(imbalance_before>lfff_flux_balance_tolerance) then
204 mean_correction=sum(bz)/dble(size(bz))
205 bz=bz-mean_correction
206 unsigned_flux=sum(dabs(bz))
207 imbalance_after=dabs(sum(bz))/max(unsigned_flux,tiny(1.d0))
208 status=1
209 end if
210 case default
211 status=2
212 end select

655 !> Vertical transfer functions for one nonzero horizontal constant-alpha
656 !> Fourier mode. transfer_b multiplies Bz0, while transfer_d=-d(transfer_b)/dz
657 !> enters the horizontal field. status is zero on success, one for an
658 !> oscillatory mode in an open half-space, two for a closed-box resonance,
659 !> and three for invalid geometry.
660 subroutine lfff_fft_transfer(k2,alpha,z,top_height,top_closed,&
661 transfer_b,transfer_d,status)
662 double precision, intent(in) :: k2,alpha,z,top_height
663 logical, intent(in) :: top_closed
664 double precision, intent(out) :: transfer_b,transfer_d
665 integer, intent(out) :: status
666
667 double precision :: q2,q,beta,denominator,scale
668
669 transfer_b=0.d0
670 transfer_d=0.d0
671 status=0
672 if(k2<=0.d0 .or. top_height<=0.d0 .or. z<0.d0 .or. z>top_height) then
673 status=3
674 return
675 end if
676
677 q2=k2-alpha**2
678 scale=max(1.d0,k2,alpha**2)
679 if(.not.top_closed) then
680 if(q2< -lfff_mode_tolerance*scale) then
681 status=1
682 return
683 end if
684 q=dsqrt(max(0.d0,q2))
685 transfer_b=dexp(-q*z)
686 transfer_d=q*transfer_b
687 else if(q2>lfff_mode_tolerance*scale) then
688 q=dsqrt(q2)
689 if(q*top_height<50.d0) then
690 denominator=dsinh(q*top_height)
691 transfer_b=dsinh(q*(top_height-z))/denominator
692 transfer_d=q*dcosh(q*(top_height-z))/denominator
693 else
694 ! Algebraically identical exponential form that cannot overflow for
695 ! large q*L when 0 <= z <= L.
696 denominator=1.d0-dexp(-2.d0*q*top_height)
697 transfer_b=(dexp(-q*z)-dexp(-q*(2.d0*top_height-z)))/denominator
698 transfer_d=q*(dexp(-q*z)+dexp(-q*(2.d0*top_height-z)))/denominator
699 end if
700 else if(q2< -lfff_mode_tolerance*scale) then
701 beta=dsqrt(-q2)
702 denominator=dsin(beta*top_height)
703 if(dabs(denominator)<lfff_resonance_tolerance) then
704 status=2
705 return
706 end if
707 transfer_b=dsin(beta*(top_height-z))/denominator
708 transfer_d=beta*dcos(beta*(top_height-z))/denominator
709 else
710 ! Critical q=0 limit of sinh(q*(L-z))/sinh(q*L).
711 transfer_b=1.d0-z/top_height
712 transfer_d=1.d0/top_height
713 end if
714 end subroutine lfff_fft_transfer

716 subroutine calc_lin_fff(ixI^L,ixO^L,Bf,x,alpha,zshift,idir)
717 ! PURPOSE:
718 ! Calculation to determine linear FFF from the field on
719 ! the lower boundary (Chiu and Hilton 1977 ApJ 212,873).
720 ! NOTE: Only works for Cartesian coordinates
721 ! INPUT: Bf,x
722 ! OUTPUT: updated b in w
723 use mod_global_parameters
724
725 integer, intent(in) :: ixI^L, ixO^L
726 integer, optional, intent(in) :: idir
727 double precision, intent(in) :: x(ixI^S,1:ndim),alpha,zshift
728 double precision, intent(inout) :: Bf(ixI^S,1:ndir)
729
730 double precision, dimension(ixO^S) :: cos_az,sin_az,zk,bigr,r,r2,r3,cos_ar,sin_ar,g,dgdz
731 double precision, dimension(ixO^S) :: dx1,dx2,invr3
732 double precision :: twopiinv
733 logical :: compute_dir(1:ndir)
734 integer :: idim,ixp1,ixp2
735
736 bf=0.d0
737 twopiinv = 0.5d0/dpi*bzmax*darea
738 zk(ixo^s)=x(ixo^s,3)-xprobmin3+zshift
739
740 compute_dir=.true.
741 if(present(idir)) then
742 compute_dir=.false.
743 if(idir>=1 .and. idir<=ndir) compute_dir(idir)=.true.
744 end if
745
746 ! For a potential field, the Green-function kernel simplifies exactly to
747 ! (dx,dy,z)/r**3. Avoid the trigonometric functions, singular 1/bigr
748 ! factors, and intermediate arrays needed by the general linear FFF form.
749 if(alpha==0.d0) then
750 do ixp2=1,nx2
751 do ixp1=1,nx1
752 dx1(ixo^s)=x(ixo^s,1)-xa1(ixp1)
753 dx2(ixo^s)=x(ixo^s,2)-xa2(ixp2)
754 r2(ixo^s)=dx1(ixo^s)**2+dx2(ixo^s)**2+zk(ixo^s)**2
755 where(r2(ixo^s)>0.d0)
756 invr3(ixo^s)=1.d0/(r2(ixo^s)*dsqrt(r2(ixo^s)))
757 elsewhere
758 invr3(ixo^s)=0.d0
759 end where
760 if(compute_dir(1)) bf(ixo^s,1)=bf(ixo^s,1)+&
761 bz0(ixp1,ixp2)*dx1(ixo^s)*invr3(ixo^s)
762 if(compute_dir(2)) bf(ixo^s,2)=bf(ixo^s,2)+&
763 bz0(ixp1,ixp2)*dx2(ixo^s)*invr3(ixo^s)
764 if(compute_dir(3)) bf(ixo^s,3)=bf(ixo^s,3)+&
765 bz0(ixp1,ixp2)*zk(ixo^s)*invr3(ixo^s)
766 end do
767 end do
768 bf(ixo^s,:)=bf(ixo^s,:)*twopiinv
769 return
770 end if
771
772 ! get cos and sin arrays for a non-zero linear force-free alpha
773 cos_az(ixo^s)=dcos(alpha*zk(ixo^s))
774 sin_az(ixo^s)=dsin(alpha*zk(ixo^s))
775 ! looping Bz0 pixels
776 do ixp2=1,nx2
777 do ixp1=1,nx1
778 bigr(ixo^s)=dsqrt((x(ixo^s,1)-xa1(ixp1))**2+&
779 (x(ixo^s,2)-xa2(ixp2))**2)
780 r2=bigr**2+zk**2
781 r=dsqrt(r2)
782 r3=r**3
783 cos_ar=dcos(alpha*r)
784 sin_ar=dsin(alpha*r)
785 where(bigr/=0.d0)
786 bigr=1.d0/bigr
787 end where
788 where(r/=0.d0)
789 r=1.d0/r
790 end where
791 where(r2/=0.d0)
792 r2=1.d0/r2
793 end where
794 where(r3/=0.d0)
795 r3=1.d0/r3
796 end where
797 g=(zk*cos_ar*r-cos_az)*bigr
798 dgdz=(cos_ar*(r-zk**2*r3)-alpha*zk**2*sin_ar*r2+alpha*sin_az)*bigr
799 do idim=1,ndim
800 if(present(idir)) then
801 if(idim/=idir) cycle
802 end if
803 select case(idim)
804 case(1)
805 bf(ixo^s,1)=bf(ixo^s,1)+bz0(ixp1,ixp2)*((x(ixo^s,1)-xa1(ixp1))*dgdz(ixo^s)&
806 +alpha*g(ixo^s)*(x(ixo^s,2)-xa2(ixp2)))*bigr(ixo^s)
807 case(2)
808 bf(ixo^s,2)=bf(ixo^s,2)+bz0(ixp1,ixp2)*((x(ixo^s,2)-xa2(ixp2))*dgdz(ixo^s)&
809 -alpha*g(ixo^s)*(x(ixo^s,1)-xa1(ixp1)))*bigr(ixo^s)
810 case(3)
811 bf(ixo^s,3)=bf(ixo^s,3)+bz0(ixp1,ixp2)*(zk(ixo^s)*cos_ar(ixo^s)*r3(ixo^s)+alpha*&
812 zk(ixo^s)*sin_ar(ixo^s)*r2(ixo^s))
813 end select
814 end do
815 end do
816 end do
817 bf(ixo^s,:)=bf(ixo^s,:)*twopiinv
818
819 end subroutine calc_lin_fff
