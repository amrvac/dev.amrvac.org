1!=============================================================================
2!> Parameters, resolved state and lifecycle of the short-characteristics (SC) radiative
3!> transfer, the Olson-Kunasz linear RT of AMRVAC
4!>
5!> The &rt_list parameters are physical, with heights in cm, pressures in dyn/cm^2 and
6!> temperatures in K. rt_sc_params_read reads them, and rt_sc_setup, run once the units are
7!> fixed, converts them to code units, snaps the top of each band to a block face, sets the
8!> horizontal boundary conditions, binds the opacity callback and builds the quadrature.
9!> rt_sc_activate, called by the physics module when hd_radiative_transfer or
10!> mhd_radiative_transfer is set, runs the reading and reserves the intensity slots that the
11!> distributed wavefront needs
12!>
13!> The module also holds the state that the solvers share. This is the ps_rt intensity state,
14!> kept out of the hydro w so that it skips the stage copies, the prolongation and the output,
15!> together with the block caches of chi and B, the KBA wave state of the current solve, the
16!> buffers of the periodic seam and the node communicators of the gather. It uses
17!> mod_rt_sc_opacity and mod_rt_sc_kernel, and mod_rt_sc_solvers uses it in turn
18!>
19!> Jack Jenkins (10.2026); More or less finished in v1 state for amrvac4.0
20!=============================================================================

175 !> Check the geometry, convert the parameters to code units, resolve the bands and the boundary
176 !> conditions, and bind rt_sc_chiB to the chosen opacity. Runs after the unit system is fixed
177 subroutine rt_sc_setup()
178 use mod_global_parameters
179 use mod_geometry, only: coordinate, cartesian
180 use mod_usr_methods, only: usr_rt_sc_chib
181 use mod_comm_lib, only: mpistop
182 integer :: ib, fac
183 integer :: dnx(ndim), bnx(ndim), mlo(ndim), mhi(ndim), glo(ndim), ghi(ndim)
184 double precision :: dlo(ndim), dhi(ndim), hi_req
185 if(.not. rt_sc_active) return
186 if(rt_sc_updim < 1) rt_sc_updim = ndim
187 ! The RT grid arithmetic assumes uniform Cartesian cells. All else is rejected for now
188 if(coordinate /= cartesian) then
189 call mpistop("rt_sc_setup: uniform Cartesian only (no stretched grids)")
190 end if
191 if(ndim /= 2) call mpistop("rt_sc_setup: 2D only")
192 if(rt_sc_updim > ndim) call mpistop("rt_sc_setup: bad rt_sc_updim")
193 ! The horizontal axis is whichever of the two is not updim
194 updim = rt_sc_updim
195 hdim = 3 - updim
196 if(rt_sc_amr_level < 1) rt_sc_amr_level = 1
197 if(rt_sc_amr_level > refine_max_level) call mpistop("rt_sc_setup: rt_sc_amr_level above refine_max_level")
198 if(rt_sc_nband < 1) rt_sc_nband = 1
199 if(rt_sc_nband > rt_sc_maxband) call mpistop("rt_sc_setup: rt_sc_nband above rt_sc_maxband")
200 ! The distributed wave keeps its intensity in slots that rt_sc_activate reserves
201 if(rt_sc_distributed .and. i_isw < 0) call mpistop("rt_sc_setup: call rt_sc_activate first")
202 ! The RT grid along (hdim, updim). The horizontal direction splits into whole blocks, since
203 ! AMRVAC requires domain_nx to be a multiple of block_nx
204 {dlo(^d)=xprobmin^d; dhi(^d)=xprobmax^d; dnx(^d)=domain_nx^d; bnx(^d)=block_nx^d;}
205 fac = 2**(rt_sc_amr_level-1)
206 rg_nh = dnx(hdim)*fac; rg_nv = dnx(updim)*fac
207 rg_bh = bnx(hdim); rg_bv = bnx(updim)
208 rg_nbh = rg_nh/rg_bh
209 rg_hlo = dlo(hdim); rg_vlo = dlo(updim)
210 rg_dh = (dhi(hdim) -dlo(hdim)) /dble(rg_nh)
211 rg_dv = (dhi(updim)-dlo(updim))/dble(rg_nv)
212 {mlo(^d)=ixmlo^d; mhi(^d)=ixmhi^d; glo(^d)=ixglo^d; ghi(^d)=ixghi^d;}
213 eh = 0; eh(hdim) = 1
214 ev = 0; ev(updim) = 1
215 imlo_h = mlo(hdim); imhi_h = mhi(hdim); iglo_h = glo(hdim); ighi_h = ghi(hdim)
216 imlo_v = mlo(updim); imhi_v = mhi(updim); iglo_v = glo(updim); ighi_v = ghi(updim)
217 ! The band edges go to code units, and the 1e30 sentinels resolve to the domain ends
218 do ib = 1, rt_sc_nband
219 band_lo_c(ib) = merge(dlo(updim), rt_sc_band_min(ib)/unit_length, rt_sc_band_min(ib) <= -1.0d29)
220 band_hi_c(ib) = merge(dhi(updim), rt_sc_band_max(ib)/unit_length, rt_sc_band_max(ib) >= 1.0d29)
221 if(band_hi_c(ib) <= band_lo_c(ib)) call mpistop("rt_sc_setup: band top not above its bottom")
222 end do
223 ! The KBA needs every band to tile into whole blocks along updim, so its top is snapped down to
224 ! the last block edge below the requested height, and everything downstream reads band_hi_c.
225 ! It never moves up, which could leave the domain or reach blocks the refinement pin has not
226 ! refined
227 do ib = 1, rt_sc_nband
228 hi_req = band_hi_c(ib)
229 ! Integer division counts only the whole blocks that fit in the requested band
230 ! nbv = floor(ncells/bv), band_hi = band_lo + nbv bv dv
231 band_nbv(ib) = nint((band_hi_c(ib)-band_lo_c(ib))/rg_dv)/rg_bv
232 if(band_nbv(ib) < 1) call mpistop("rt_sc_setup: band thinner than one block along updim")
233 band_nv(ib) = band_nbv(ib)*rg_bv
234 band_hi_c(ib) = band_lo_c(ib) + dble(band_nv(ib))*rg_dv
235 if(mype==0 .and. dabs(hi_req-band_hi_c(ib)) > 1.0d-10*dabs(rg_dv)) then
236 write(*,'(a,i2,a,es13.6,a,es13.6,a,i0,a,i0,a)') &
237 ' rt_sc_setup: band ', ib, ' top snapped ', hi_req*unit_length, &
238 ' -> ', band_hi_c(ib)*unit_length, ' cm (', band_nbv(ib), ' blocks of ', rg_bv, ' cells)'
239 end if
240 end do
241 ! The switch pressure is held as a code-unit logarithm, so the cells need no conversion. The
242 ! opacity callback is bound once here, and the table is read only when it is needed
243 rt_sc_lp_tr = dlog10(rt_sc_p_tr/unit_pressure)
244 select case(trim(rt_sc_opacity))
245 case('Rosseland')
246 call read_opacity_table()
247 rt_sc_chib => rt_sc_chib_rosseland
248 case('Kramers')
249 rt_sc_chib => rt_sc_chib_kramers
250 case('usr')
251 if(.not. associated(usr_rt_sc_chib)) &
252 call mpistop("rt_sc_setup: rt_sc_opacity='usr' but usr_rt_sc_chiB not set in mod_usr.t")
253 rt_sc_chib => usr_rt_sc_chib
254 case default
255 call mpistop("rt_sc_setup: unknown rt_sc_opacity (Rosseland, Kramers or usr)")
256 end select
257 ! The horizontal edges wrap only where the domain is periodic, and otherwise take the inflow
258 ! conditions given in &rt_list
259 bc_h_periodic = periodb(hdim)
260 bc_hlo_thermal = (trim(rt_sc_bc_hmin) == 'thermal')
261 bc_hhi_thermal = (trim(rt_sc_bc_hmax) == 'thermal')
262 if(mype==0 .and. .not.bc_h_periodic) print *, &
263 'rt_sc_setup: hdim not periodic, inflow BC ', trim(rt_sc_bc_hmin),'/',trim(rt_sc_bc_hmax)
264
265 ! The GL polar nodes are placed about the stratification axis, and the cosines are then mapped
266 ! onto the RT grid axes
267 rt_sc_polar_axis = updim
268 call rt_sc_init()
269 call rt_sc_axis_cosines()
270 rt_sc_ready = .true.
271 if(mype==0) print *, "rt_sc_setup: done. updim=",updim," level=",rt_sc_amr_level, &
272 " nband=",rt_sc_nband," opacity=",trim(rt_sc_opacity)
273 end subroutine rt_sc_setup

306 !> Allocate ps_rt(igrid)%w on every active block and zero it, sharing the geometry of ps, which
307 !> getbc reads. With bcphys off only %w is exchanged. Blocks created by a regrid get their state
308 !> at the next solve, and blocks that go inactive keep it until they are reused
309 subroutine rt_sc_alloc_state()
310 use mod_global_parameters
311 integer :: iigrid, igrid
312 if(.not.allocated(ps_rt)) allocate(ps_rt(max_blocks))
313 do iigrid=1,igridstail; igrid=igrids(iigrid)
314 ! A block whose state was sized for a different nw is allocated afresh
315 if(allocated(ps_rt(igrid)%w)) then
316 if(size(ps_rt(igrid)%w,ndim+1) /= nw) deallocate(ps_rt(igrid)%w)
317 end if
318 if(.not.allocated(ps_rt(igrid)%w)) allocate(ps_rt(igrid)%w(ixg^t,1:nw))
319 ! The state is zeroed every solve, because getbc with bcphys off never fills the
320 ! physical-boundary ghosts, which would otherwise keep stale values
321 ps_rt(igrid)%w = 0.0d0
322 ps_rt(igrid)%igrid = igrid
323 ps_rt(igrid)%level = node(plevel_,igrid)
324 ! The geometry points at the hydro block rather than being copied
325 {ps_rt(igrid)%ixGmin^d = ixglo^d; ps_rt(igrid)%ixGmax^d = ixghi^d;}
326 ps_rt(igrid)%x => ps(igrid)%x
327 ps_rt(igrid)%dvolume => ps(igrid)%dvolume
328 ps_rt(igrid)%is_physical_boundary=> ps(igrid)%is_physical_boundary
329 end do
330 end subroutine rt_sc_alloc_state

361 !> In-place sum allreduce of a 2D array in chunks of whole columns of about 1 MB
362 subroutine rt_allreduce_chunk(a, n1, n2, comm)
363 use mod_global_parameters
364 integer, intent(in) :: n1, n2, comm
365 double precision, intent(inout) :: a(n1,n2)
366 integer :: j0, j1, nc, ierr
