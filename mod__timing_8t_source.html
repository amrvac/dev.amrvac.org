43 !> Open an EoS timing window; inside an OpenMP parallel region the EoS is not timed
44 subroutine eos_timer_start()
45 !$ if(omp_in_parallel()) return
46 if(eos_depth==0) eos_t0=mpi_wtime()
47 eos_depth=eos_depth+1
48 end subroutine eos_timer_start
49
50 !> Close an EoS timing window, adding its time to acc when it is the outermost
51 subroutine eos_timer_stop(acc)
52 double precision, intent(inout) :: acc
53 !$ if(omp_in_parallel()) return
54 eos_depth=eos_depth-1
55 if(eos_depth==0) acc=acc+(mpi_wtime()-eos_t0)
56 end subroutine eos_timer_stop

63 !> Open a timing window, whose length leaves out the EoS and ghost-cell time spent inside it,

66 use mod_global_parameters, only: time_bc
