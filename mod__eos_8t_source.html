1!> Equation of state for AMRVAC, handled through a single eos_container object.
2!>
3!> Two gas models (eos_type):
4!> FI -- fully ionised ideal gas (constant gamma, mu).
5!> LTE -- local thermodynamic equilibrium with partial ionisation of H (+He),
6!> via one of three interchangeable methods (eos_method):
7!> state -- per-quantity (T,p,ne/nH,eint/p) Saha-equilibrium tables,
8!> PCHIP/bilinear interpolated ('tables' accepted as legacy alias).
9!> entropy -- bicubic-Hermite reconstruction (see mod_eos_lte_entropy).
10!> analytic -- on-the-fly hydrogen Saha solve, helium neutral (saha_* routines).
11!> Tables are generated offline and read as binary; their axes are
12!> (log10 nH, log10 eint/nH) in code units after the unit shift applied
13!> in eos_finalise. ionE optionally folds ionisation energy into eint.
14!>
15!> The code-unit normalisation is kept so an FI atmosphere recovers R = 1;
16!> update_eos_LTE relies on this. update_eos refreshes the cached EoS fields
17!> (Te, ne) once per RK substep, and for boundary cells via update_eos_4_bc;
18!> call eos%update_eos manually before any extra output.
19!>
20!> Jack Jenkins (27.01.2026); legacy EoS type 1: uniform tabulated (by Chris Osborne) H(+He) LTE EoS, in
21!> consultation with Damien Przybylski (MURaM). Routed through legacy usr_Rfactor approach.
22!> Jack Jenkins (05.05.2026); EoS type 2 finished, now separate module that each hd/mhd physics module
23!> from upstream/amrvac3.3 route through.
24!> Jack Jenkins (11.05.2026); Benchmarked curvature-density-hybrid-tabulated EoS tables.
25!> Jack Jenkins (25.05.2026); Single curvature-density-tabulated entropy potential EoS prototyped. Validated
26!> gamma(T) against Ballester et al. 2001 to ~ 1.1% .
27!> Jack Jenkins (16.06.2026); Folded in fld, ffhd, and new partial_ionisation modules to be EoS compliant
28!> and refactored for readability.

85 !> Namelist mirror variables: the parfile uses these simple names
86 !> (e.g. eos_type='FI'); they are committed to eos% in one block after
87 !> the read. Derived eos% quantities (gamma_minus_1, method_id, ...) are
88 !> NOT mirrored here -- they are computed from these inputs below.
89 character(len=std_len) :: eos_type, table_location
90 character(len=20) :: eos_method, gamma1_method, p2eint_method, pi_table
91 double precision :: He_abundance, gamma
92 logical :: ionE

94 namelist /eos_list/ eos_type, table_location, ione, he_abundance, gamma, &
95 eos_method, gamma1_method, p2eint_method, pi_table
96
97 !> defaults
98 eos_type = 'FI'
99 eos_method = 'state'
100 pi_table = 'chromosphere'
101 gamma1_method = 'exact'
102 p2eint_method = 'table'
103 ione = .false.
104 he_abundance = 0.1d0
105 gamma = 5.0d0/3.0d0
106 call get_environment_variable("AMRVAC_DIR", amrvac_dir)
107 table_location = trim(amrvac_dir)//"/src/tables/eos_tables/uniform256/"
108
109 !> read parfile(s): each successive file overrides the previous
110 do n = 1, size(files)
111 open(unitpar, file=trim(files(n)), status="old")
112 read(unitpar, eos_list, end=111)
113 111 close(unitpar)
114 end do

116 !> commit raw inputs to eos% (plain copies, no logic)
117 eos%eos_type = eos_type
118 eos%method = eos_method
119 eos%pi_table = pi_table
120 eos%gamma1_method = gamma1_method
121 eos%p2eint_method = p2eint_method
122 eos%table_location = table_location
123 eos%He_abundance = he_abundance
124 eos%gamma = gamma
125 eos%ionE = ione
126
127 !> derive secondary quantities from the raw inputs
128 eos%gamma_minus_1 = eos%gamma - 1.0d0
129 eos%inv_gamma = 1.0d0 / eos%gamma
130 if (abs(eos%gamma_minus_1) > 1.0d-12) then
131 eos%inv_gamma_minus_1 = 1.0d0 / eos%gamma_minus_1
132 else
133 eos%inv_gamma_minus_1 = 0.0d0
134 end if
135 eos%nH2rhoFactor = 1.0d0 !> FI default; (m)hd_physical_units may reset
136 select case (eos%method)
137 case ('analytic'); eos%method_id = eos_analytic
138 case ('entropy'); eos%method_id = eos_entropy
139 case default; eos%method_id = eos_state
140 end select
141 select case (eos%eos_type)
142 case ('LTE'); eos%type_id = eos_type_lte
143 case ('PI'); eos%type_id = eos_type_pi
144 case default; eos%type_id = eos_type_fi
145 end select
146
147 !> report
148 if (mype == 0) then
149 write(*,*) "EoS type: "//trim(eos%eos_type)
150 if (eos%eos_type == 'LTE') &
151 write(*,*) "EoS table location: "//trim(eos%table_location)
152 end if

154 !> validate / coerce (all consistency guards live in one place)
155 call eos_validate_params()

159 !> Validate and coerce the parsed eos_list against the chosen eos_type /
160 !> method. Grouped by concern so the rules are readable in one block (rather
161 !> than interleaved with the field copies in eos_read_params). May coerce
162 !> eos% fields (e.g. FI forces ionE=.false.) and mpistop on illegal combos.
163 subroutine eos_validate_params()
164
165 !> eos_type x ionE
166 if (eos%eos_type == 'FI' .and. eos%ionE) then
167 eos%ionE = .false.
168 if(mype==0) write(*,*) "WARNING: eos_type = 'FI' requires ionE = .false."
169 end if
170
171 !> PI ('partial approximations') table selector
172 if (eos%eos_type == 'PI') then
173 select case (trim(eos%pi_table))
174 case ('chromosphere','flare','prominence')
175 ! valid
176 case default
177 call mpistop("eos_type='PI': pi_table must be "// &
178 "'chromosphere', 'flare', or 'prominence'")
179 end select
180 !> ionisation-energy mode (ionE) is supported only for the T-only
181 !> tables; the prominence table is (T,p)-dependent -> no-energy only.
182 !> This also gates the only PI configuration that cannot model He
183 !> ionisation energy: the chromosphere/flare energy paths now fold
184 !> the He ionisation energy into eint (mod_eos_pi_tables,
185 !> ionization_eps_ion_of_degrees), consistent with the He electron
186 !> count already in the R-factor, so He>0 energy mode is allowed there.
187 if (eos%ionE .and. trim(eos%pi_table) == 'prominence') then
188 call mpistop("eos_type='PI' with ionE=T requires a T-only table "// &
189 "(chromosphere or flare); prominence is no-energy only")
190 end if
191 if(mype==0) write(*,*) "EoS PI table: "//trim(eos%pi_table)// &
192 ", ionE=", eos%ionE
193 end if
194
195 !> eos_method validity and eos_type x method ('tables' kept as a legacy
196 !> alias for 'state' -- both decode to EOS_STATE via the case default).
197 if (eos%method /= 'analytic' .and. eos%method /= 'state' &
198 .and. eos%method /= 'tables' .and. eos%method_id /= eos_entropy) then
199 call mpistop("Unknown eos_method: "//trim(eos%method)// &
200 " (expected 'state', 'analytic', or 'entropy')")
201 end if
202 if (eos%eos_type == 'FI' .and. eos%method_id == eos_analytic) then
203 call mpistop("eos_method='analytic' requires eos_type='LTE'")
204 end if
205
206 !> gamma1_method validity
207 if (eos%gamma1_method /= 'exact' .and. eos%gamma1_method /= 'effective' &
208 .and. eos%gamma1_method /= 'constant') then
209 call mpistop("Unknown gamma1_method: "//trim(eos%gamma1_method)// &
210 " (expected 'exact', 'effective', or 'constant')")
211 end if
212
213 !> analytic-method caveats (hydrogen Saha, helium neutral)
214 if (eos%method_id == eos_analytic) then
215 if (eos%He_abundance > 0.0d0 .and. mype == 0) write(*,*) &
216 "eos_method='analytic' keeps helium neutral, He_abundance=", eos%He_abundance
217 ! effective gamma1 needs T,neOnH tables -> not available analytically
218 if (eos%gamma1_method == 'effective') then
219 if (mype == 0) write(*,*) &
220 "WARNING: gamma1_method='effective' not applicable with analytic EoS."
221 if (mype == 0) write(*,*) &
222 " Falling back to gamma1_method='exact' (analytic 2D table)."
223 eos%gamma1_method = 'exact'
224 end if
225 if (.not. eos%ionE .and. mype == 0) then
226 write(*,*) "WARNING: eos_method='analytic' without ionE"
227 write(*,*) " reduces to ideal gas. Consider eos_type='FI'."
228 end if
229 end if
230
231 !> LTE+ionE effective-gamma1 note
232 if (eos%eos_type == 'LTE' .and. eos%ionE .and. &
233 eos%gamma1_method == 'effective') then
234 if (mype == 0) write(*,*) &
235 "NOTE: gamma1_method='effective' builds gamma_eff = 1+p/eint table."
236 end if
237
238 end subroutine eos_validate_params

240 !> Phase 'create' (before units are known): allocate the EoS object, read
241 !> its &eos_list parameters, and load the raw tables for the chosen method.
242 !> The getters wired here must be live before (m)hd_link_eos runs.
243 subroutine eos_init()
244 allocate(eos)
245 call eos_read_params(par_files)
246
247 !> Type-agnostic getters: always live (eos% pointer targets).
248 eos%get_rho => get_rho
249 eos%get_nH => get_nh
250 eos%get_ne_nH => get_ne_nh

252 !> Per-type create-phase init (each owns its raw table loads / pointer
253 !> wiring that must be live before (m)hd_link_eos runs), in the matching
254 !> mod_eos_<TYPE> module.
255 select case (eos%eos_type)
256 case ('FI'); call eos_init_fi()
257 case ('LTE'); call eos_init_lte()
258 case ('PI'); call eos_init_pi()
259 case default
260 call mpistop('eos_init: eos_type '//trim(eos%eos_type)//' not recognised')
261 end select
262 end subroutine eos_init
263
264 !> Phase 'commit' (after units are known): finalise the dispatch for the
265 !> loaded physics -- wire the method's runtime pointers and finalise its
266 !> tables. The cross-module wiring that follows (into the ghost-cell update
267 !> and the physics source terms) is done by the caller in amrvac.t, so the
268 !> EoS never reaches back into those modules.
269 !> Finalise the EoS once units are set: dispatch on eos_type to the per-type
270 !> finaliser (each owns its pointer wiring + backend table loading, living in
271 !> mod_eos_fi / mod_eos_lte / mod_eos_pi), then wire the shared total-energy
272 !> temperature getter. Called from amrvac.t after phys init.
273 subroutine eos_finalise()
274 if (.not. allocated(eos)) &
275 call mpistop('eos_finalise: eos not allocated')
276
277 ! Decoded here, not at read time: physics init may still override the
278 ! method (hd well-balanced forces 'bisect')
279 eos%p2eint_bisect = (eos%p2eint_method == 'bisect')
280
281 select case (eos%eos_type)
282 case ('LTE')
283 call eos_finalise_lte()
284 case ('FI')
285 call eos_finalise_fi()
286 case ('PI')
287 call eos_finalise_pi()
288 case default
289 call mpistop('eos_finalise: eos_type '//trim(eos%eos_type)//' not recognised')
290 end select
291
292 eos%get_temperature_from_etot => get_temperature_from_etot
293 ! The eint/nH floor of every LTE path: the forward-table edge, or 100 K of neutral gas
294 ! under analytic, which has no tables
295 if (eos%type_id == eos_type_lte) then
296 if (eos%method_id == eos_analytic) then
297 eos%eint_nH_min = eos%inv_gamma_minus_1 * 100.0d0 / unit_temperature
298 else
299 eos%eint_nH_min = 10.0d0**eos%T%var2_min
300 end if
301 end if
302 call eos_time_dispatch()
303 end subroutine eos_finalise
