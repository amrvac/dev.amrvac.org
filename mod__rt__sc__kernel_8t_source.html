1!=============================================================================
2!> Transport kernel of the short-characteristics (SC) radiative transfer
3!>
4!> The transfer equation is solved along one ray direction at a time with the linear short
5!> characteristics of Olson and Kunasz. Each cell takes the intensity at the upwind point on the
6!> face the ray crosses first, interpolated linearly along that face, and integrates across one
7!> step with the source taken as linear in optical depth. sc_sweep_block sweeps one block whose
8!> ghost ring already holds the upwind values and the inflows, and it is the only sweep in the
9!> code. The wavefront solvers call it block by block, and the gather calls it once on the whole
10!> replicated band, which sc_pad_grid has padded with the inflows
11!>
12!> The module also builds the angular quadrature, Gauss-Legendre or the Carlson A4 set, and in 2D
13!> merges the rays that differ only out of the plane. It holds the replicated single-level grid
14!> of the gather and forms the heating in blend_heating, which blends c chi (J - B) with -div F.
15!> Nothing here communicates, since the drivers fill chi and B beforehand
16!>
17!> The gather is the original implementation, kept for comparison and not for production. Every
18!> rank holds the whole band, and its periodic seam is not exact (see sc_sweep_2d). Production runs
19!> should use KBA, which is far better optimised
20!>
21!> Structure:
22!> A. quadrature rt_sc_init, rt_sc_dedup_rays, build_gl, build_carlson and
23!> gauss_legendre_nodes;
24!> B. heating blend_heating, and rt_sc_heat_blend on the replicated grid;
25!> C. transport sc_pad_grid and sc_sweep_2d for the replicated grid, ka_coeffs for the
26!> weights of one step, and sc_sweep_block
27!>
28!> Jack Jenkins (10.2026); More or less finished in v1 state for amrvac4.0
29!=============================================================================

78 !> Build the quadrature into mux, muy, muz and wmu as full 3D cosines, according to rt_sc_quad
79 subroutine rt_sc_init()
80 use mod_global_parameters
81 ! Even in 2D we build the full 3D set, because the out-of-plane cosine still sets how long
82 ! each in-plane step is
83 select case(trim(adjustl(rt_sc_quad)))
84 case('Carlson','carlson')
85 call build_carlson()
86 case default
87 call build_gl()
88 end select
89 ! Rays that differ only in muz are indistinguishable to a 2D sweep, so we merge them and
90 ! save the repeated work
91 call rt_sc_dedup_rays()
92 if(mype==0) print *, "rt_sc_init: quadrature '"//trim(adjustl(rt_sc_quad))//"', rays =", rt_sc_nray
93 end subroutine rt_sc_init

95 !> Merge rays that share (mux, muy) and add up their weights. Since the 2D sweep and the moments
96 !> see only those two cosines, Carlson's +-muz pairs and GL's phi, 2pi-phi pairs sweep the same.
97 !> The total weight is preserved, so J, Fx and Fy are unchanged to round-off
98 subroutine rt_sc_dedup_rays()
99 use mod_global_parameters
100 double precision, allocatable :: nx(:), ny(:), nz(:), wsum(:)
101 logical, allocatable :: gone(:)
102 integer :: k, j, nnew
103 ! Cosines that agree to this tolerance are treated as the same direction
104 double precision, parameter :: tol = 1.0d-12
105 if(rt_sc_nray <= 1) return
106 allocate(nx(rt_sc_nray), ny(rt_sc_nray), nz(rt_sc_nray), wsum(rt_sc_nray), gone(rt_sc_nray))
107 gone = .false.; nnew = 0
108 ! Each ray not yet absorbed opens a new entry in the merged set, and any later ray with the
109 ! same in-plane cosines hands its weight over and is marked gone
110 do k = 1, rt_sc_nray
111 if(gone(k)) cycle
112 nnew = nnew + 1
113 nx(nnew) = mux(k); ny(nnew) = muy(k); nz(nnew) = muz(k); wsum(nnew) = wmu(k)
114 ! w_merged = w_k + the sum of w_j over the later rays j with the same (mux, muy)
115 do j = k+1, rt_sc_nray
116 if(gone(j)) cycle
117 if(abs(mux(j)-mux(k)) < tol .and. abs(muy(j)-muy(k)) < tol) then
118 wsum(nnew) = wsum(nnew) + wmu(j); gone(j) = .true.
119 end if
120 end do
121 end do
122 ! The quadrature is only replaced when something was actually merged
123 if(nnew < rt_sc_nray) then
124 deallocate(mux, muy, muz, wmu)
125 allocate(mux(nnew), muy(nnew), muz(nnew), wmu(nnew))
126 mux = nx(1:nnew); muy = ny(1:nnew); muz = nz(1:nnew); wmu = wsum(1:nnew)
127 if(mype==0) print *, "rt_sc_dedup_rays: (mux,muy) rays", rt_sc_nray, "->", nnew, " (2D-identical merged)"
128 rt_sc_nray = nnew
129 end if
130 deallocate(nx, ny, nz, wsum, gone)
131 end subroutine rt_sc_dedup_rays

133 !> Product Gauss-Legendre set, with nmu polar nodes about rt_sc_polar_axis and ndir azimuths at
134 !> the midpoints of equal intervals
135 subroutine build_gl()
136 use mod_global_parameters
137 integer :: im, ip, k, nmu, nphi
138 double precision :: phi, s, mu, xmu(8), wgl(8)
139 ! With fewer than three azimuths the plane is not resolved, so we never go below three
140 if(rt_sc_ndir < 3) rt_sc_ndir = 3
141 nphi = rt_sc_ndir; nmu = rt_sc_nmu
142 ! The node routine may round nmu up to the next order it supports
143 call gauss_legendre_nodes(nmu, xmu, wgl)
144 rt_sc_nray = nmu*nphi
145 if(allocated(mux)) deallocate(mux, muy, muz, wmu)
146 allocate(mux(rt_sc_nray), muy(rt_sc_nray), muz(rt_sc_nray), wmu(rt_sc_nray))
147 k = 0
148 do im = 1, nmu
149 ! The polar cosine and its sine, mu = x_GL(im), s = sqrt(1 - mu^2)
150 mu = xmu(im)
151 s = dsqrt(max(1.0d0 - mu*mu, 0.0d0))
152 do ip = 1, nphi
153 ! phi = 2 pi (ip - 1/2)/nphi, mu_h = s cos(phi), mu_z = s sin(phi)
154 phi = 2.0d0*dpi*(dble(ip)-0.5d0)/dble(nphi)
155 k = k + 1
156 ! The polar cosine lies along the stratification and the azimuth turns about it
157 if(rt_sc_polar_axis == 2) then
158 muy(k) = mu; mux(k) = s*dcos(phi)
159 else
160 mux(k) = mu; muy(k) = s*dcos(phi)
161 end if
162 muz(k) = s*dsin(phi)
163 ! The GL weights add up to 2 on [-1, 1] and the azimuths share each one equally
164 ! w = w_GL/(2 nphi), so that sum w = 1
165 wmu(k) = 0.5d0*wgl(im)/dble(nphi)
166 end do
167 end do
168 end subroutine build_gl

170 !> Carlson A4 level-symmetric set, built from the three cyclic permutations of (mu1, mu2, mu2)
171 !> with mu1^2 + 2 mu2^2 = 1 in each of the 8 octants, which gives 24 directions of weight 1/24
172 !> as in MURaM's carlson3
173 subroutine build_carlson()
174 ! m1^2 + 2 m2^2 = 1, so every direction built from these is a unit vector
175 double precision, parameter :: m1 = 0.88191710d0, m2 = 0.33333333d0
176 double precision :: base(3,3)
177 integer :: ib, sgx, sgy, sgz, k
178 ! Each column places the long cosine on a different axis
179 base(:,1) = (/ m1, m2, m2 /)
180 base(:,2) = (/ m2, m1, m2 /)
181 base(:,3) = (/ m2, m2, m1 /)
182 rt_sc_nray = 24
183 if(allocated(mux)) deallocate(mux, muy, muz, wmu)
184 allocate(mux(24), muy(24), muz(24), wmu(24))
185 k = 0
186 ! The sign loops run over the eight octants, and every octant receives all three
187 ! permutations with the same weight, as a level-symmetric set requires
188 do sgz = -1, 1, 2; do sgy = -1, 1, 2; do sgx = -1, 1, 2
189 do ib = 1, 3
190 k = k + 1
191 ! mu = (sgx b_1, sgy b_2, sgz b_3) for the permutation b, w = 1/24
192 mux(k) = sgx*base(1,ib); muy(k) = sgy*base(2,ib); muz(k) = sgz*base(3,ib)
193 wmu(k) = 1.0d0/24.0d0
194 end do
195 end do; end do; end do
196 end subroutine build_carlson

231 !> Heating of one cell. The thin imbalance c chi (J - B) is blended with the thick, conservative
232 !> -div F through w = exp(-rt_sc_blend_k tau), and the taper tau^2/(tau^2 + tau_min^2) removes
233 !> the heating where the column above is optically thin
234 pure double precision function blend_heating(chi, jm, bpl, divf, tau) result(q)
235 use mod_global_parameters, only: c_norm
236 double precision, intent(in) :: chi, jm, bpl, divf, tau
237 double precision :: w
238 ! w = exp(-k tau)
239 ! Q = [w c chi (J - B) + (1 - w)(-div F)] tau^2/(tau^2 + tau_min^2)
240 w = dexp(-rt_sc_blend_k*tau)
241 q = (w*(c_norm*chi*(jm-bpl)) + (1.0d0-w)*(-divf)) * tau*tau/(tau*tau + rt_sc_tau_min**2)
242 end function blend_heating
243
244 !> Heating q_g, blending the thin imbalance c chi (J - B) with the thick, conservative -div F
245 !> through w = exp(-rt_sc_blend_k tau), tau measured from the top. Taking -div F at depth avoids
246 !> the stiff, cancelling J - B there. Needs jm_g, b_g, fx_g, fy_g and chi_g
247 subroutine rt_sc_heat_blend()
248 use mod_global_parameters
249 integer :: i, j, ip, im
250 double precision :: divf
251 ! The blend and the taper both depend on how far below the top each cell sits in optical
252 ! depth, so we integrate tau down every column first. The top cell centre lies half a cell
253 ! below the band top, and each step down adds the trapezoid of chi between two centres
254 do i = 1, nxg
255 ! tau_top = chi_top dy/2, tau_j = tau_(j+1) + (chi_(j+1) + chi_j) dy/2
256 tau_g(i,nyg) = 0.5d0*chi_g(i,nyg)*dyg
257 do j = nyg-1, 1, -1
258 tau_g(i,j) = tau_g(i,j+1) + 0.5d0*(chi_g(i,j+1)+chi_g(i,j))*dyg
259 end do
260 end do
261 do j = 1, nyg
262 do i = 1, nxg
263 ! The flux divergence uses centred differences, wrapping round in x when the domain is
264 ! periodic and falling back to one-sided differences at a closed edge or a band end.
265 ! This latter point introduces non negligible errors since we do not illuminate from the
266 ! sides...
267 ! div F = (Fx_(i+1) - Fx_(i-1))/(2 dx) + (Fy_(j+1) - Fy_(j-1))/(2 dy)
268 if(bc_h_periodic) then
269 ip = i+1; if(ip>nxg) ip=1
270 im = i-1; if(im<1) im=nxg
271 divf = (fx_g(ip,j)-fx_g(im,j))/(2.0d0*dxg)
272 else
273 ip = min(i+1,nxg); im = max(i-1,1)
274 divf = (fx_g(ip,j)-fx_g(im,j))/(dble(ip-im)*dxg)
275 end if
276 if(j==1) then
277 divf = divf + (fy_g(i,2)-fy_g(i,1))/dyg
278 else if(j==nyg) then
279 divf = divf + (fy_g(i,nyg)-fy_g(i,nyg-1))/dyg
280 else
281 divf = divf + (fy_g(i,j+1)-fy_g(i,j-1))/(2.0d0*dyg)
282 end if
283 q_g(i,j) = blend_heating(chi_g(i,j), jm_g(i,j), b_g(i,j), divf, tau_g(i,j))
284 end do
285 end do
286 end subroutine rt_sc_heat_blend

298 !> Pad the replicated grid with one ghost layer for sc_sweep_block, once per band after chi_g
299 !> and b_g are filled. Beyond the band ends chi and B take the edge rows and the intensity is

339 !> Textbook Olson-Kunasz weights for linear short characteristics over an optical-depth step dtau
340 subroutine ka_coeffs(dtau, ex, psi0, psiu)
341 double precision, intent(in) :: dtau
342 double precision, intent(out) :: ex, psi0, psiu
343 double precision :: w0, w1
344 ! A step this thin is transparent, so the intensity passes through unchanged
345 if(dtau <= 1.0d-10) then
346 ex = 1.0d0; psi0 = 0.0d0; psiu = 0.0d0; return
347 end if
348 ! The upwind intensity is attenuated by exp(-dtau). The source contributes the integrals of
349 ! e^-t and t e^-t over the step, which we evaluate in closed form unless dtau is so small that
350 ! the closed forms cancel, in which case their Taylor series are used
351 ! I = I_u exp(-dtau) + psi_u S_u + psi_0 S_0, with
352 ! w0 = int_0^dtau exp(-t) dt = 1 - exp(-dtau)
353 ! w1 = int_0^dtau t exp(-t) dt = w0 - dtau exp(-dtau)
354 ex = dexp(-dtau)
355 if(dtau > 1.0d-4) then
356 w0 = 1.0d0 - ex
357 w1 = w0 - dtau*ex
358 else
359 ! w0 = dtau - dtau^2/2 + dtau^3/6, w1 = dtau^2/2 - dtau^3/3
360 w0 = dtau - dtau*dtau*0.5d0 + dtau*dtau*dtau/6.0d0
361 w1 = dtau*dtau*0.5d0 - dtau*dtau*dtau/3.0d0
362 end if
363 ! With the source varying linearly in tau, w0 divides into a weight on the upwind source and a
364 ! weight on the local one
365 ! psi_u = w1/dtau, psi_0 = w0 - psi_u
366 psiu = w1/dtau
367 psi0 = w0 - psiu
368 end subroutine ka_coeffs

370 !> Sweep of one block for ray (nx, ny), the kernel of the wavefront solvers. The upwind face and
371 !> corner values sit in the ghost ring that getbc fills. With xwrap the block spans the whole
372 !> periodic x-domain and wraps iu in place. Otherwise the x-upwind values come from the ghosts,
373 !> and the block at the periodic seam reads the wrap ghost left by the previous solve
374 subroutine sc_sweep_block(nx, ny, dxb, dyb, xwrap, ixI^L, ixO^L, chi, bpl, ii)
375 use mod_global_parameters
376 double precision, intent(in) :: nx, ny, dxb, dyb
377 logical, intent(in) :: xwrap
378 integer, intent(in) :: ixI^L, ixO^L
379 double precision, intent(in) :: chi(ixI^S), bpl(ixI^S)
380 double precision, intent(inout) :: ii(ixI^S)
381 integer :: i, j, sx, sy, nxper
382 double precision :: absnx, absny, ds, frac
383 absnx = dabs(nx); absny = dabs(ny)
384 sx = ray_sign(nx); sy = ray_sign(ny)
385 ! The block width in cells, needed when the block wraps onto itself in x
386 nxper = ixomax1 - ixomin1 + 1
387 ! The upwind point lies on the face the ray crosses first, the x-face when
388 ! abs(mu_x) dy >= abs(mu_y) dx. Its first neighbour A is then the x-neighbour (iu, j) and the
389 ! columns are swept in turn, and otherwise A is the y-neighbour (i, ju) and the rows are
390 ! swept in turn. The second neighbour B is always the diagonal (iu, ju)
391 if(absnx*dyb >= absny*dxb) then
392 ! ds = dx/abs(mu_x), frac = (dx/dy) abs(mu_y)/abs(mu_x)

433 !> One cell (i, j) from its upwind neighbours A = (ia, ja) and B = (iu, ju). The upwind
434 !> values may sit in ghost cells, where the driver has already written the band-end and edge
435 !> inflows. chi and B do not depend on the ray, so their ghosts are always current
436 subroutine step(i, j, ia, ja, iu, ju)
437 integer, intent(in) :: i, j, ia, ja, iu, ju
438 double precision :: in_a, in_b, iu_val, chiu, bu, dtau, ex, psi0, psiu
439 in_a = ii(ia,ja)
440 ! The intensity in the diagonal ghost is less reliable. In the pipelined solver it may
441 ! already belong to the next ray, while both face ghosts still hold the current. Naturally
442 ! this is problematic so we avoid it by reconstructing it from the adjacent faces instead.
443 ! Seems to be a reasonable approximation at least
444 if( (iu<ixomin1 .or. iu>ixomax1) .and. (ju<ixomin2 .or. ju>ixomax2) ) then
445 ! I_B = (I(iu,j) + I(i,ju))/2
446 in_b = 0.5d0*(ii(iu,j) + ii(i,ju))
447 else
448 in_b = ii(iu,ju)
449 end if
450 ! I_u = (1 - frac) I_A + frac I_B, and chi_u and B_u alike
451 iu_val = (1.0d0-frac)*in_a + frac*in_b
452 chiu = (1.0d0-frac)*chi(ia,ja) + frac*chi(iu,ju)
453 bu = (1.0d0-frac)*bpl(ia,ja) + frac*bpl(iu,ju)
454 ! dtau = (chi_u + chi) ds/2, I = I_u exp(-dtau) + B psi_0 + B_u psi_u
455 dtau = 0.5d0*(chiu + chi(i,j))*ds
456 call ka_coeffs(dtau, ex, psi0, psiu)
457 ii(i,j) = iu_val*ex + bpl(i,j)*psi0 + bu*psiu
458 end subroutine step
459 end subroutine sc_sweep_block
