1!=============================================================================
2!> Solvers of the short-characteristics (SC) radiative transfer and their dispatch
3!>
4!> rt_sc_solve runs once per step. It sweeps every ray through every band on rt_sc_amr_level and
5!> leaves the heating Q in i_qrad and the vertical flux Fy in i_frad. The pipelined KBA wavefront
6!> is the default. Its sweep corners run at once, each in its own intensity slot, and the rays of
7!> a corner follow one another through the wavefront. The ray-by-ray wavefront passes one ray at
8!> a time, agrees with it to round-off, and is the simplest to read. In both, each rank
9!> sweeps only its own band blocks, one diagonal of blocks at a time, with a getbc of the ray
10!> intensity between diagonals, so that no rank holds the whole band. The replicated gather
11!> assembles chi and B of the whole band on every rank and is kept as the reference. It differs
12!> from the wavefront only where rays cross a periodic seam
13!>
14!> The loops over block cells are written along (hdim, updim) through hv, so that one loop
15!> serves either stratification axis
16!>
17!> Structure:
18!> A. dispatch rt_sc_solve and the validation rt_sc_validate_pipe;
19!> B. gather rt_sc_solve_gather and rt_grid_index;
20!> C. band blocks rt_band_blocks, which lists the band blocks and caches chi and B, with
21!> rt_tau_tops and rt_block_tau for tau from the band top, which
22!> rt_sc_emergent_tau1 shares;
23!> D. wave stages rt_wave_begin, rt_wave_band_open, rt_wave_band_close, which forms Q, and
24!> rt_wave_end, with rt_add_moments and rt_exchange;
25!> E. block work rt_seam_read and rt_seam_write for the periodic seam, rt_inflow_ghosts for
26!> the band-end and edge inflows, and rt_sweep_one;
27!> F. schedules rt_sc_solve_wave and rt_sc_solve_wave_pipe, with wave_stage
28!>
29!> Jack Jenkins (10.2026); More or less finished in v1 state for amrvac4.0
30!=============================================================================

703 !> Pipelined KBA. The direction-sign corners run together in batches of nisw slots, and within
704 !> each corner the rays follow one another through its wavefront. The result agrees with that
705 !> of rt_sc_solve_wave to round-off, since the moments are summed in another order
706 subroutine rt_sc_solve_wave_pipe()
707 use mod_global_parameters
708 use mod_ghostcells_update, only: create_bc_mpi_datatype
709 integer :: ib, k, sx, sy, iblk, igrid, islot, nslot
710 integer :: ng, gidx, g, jj, tt_clk, tmax, gbase, gend, gg
711 integer :: grp_sx(9), grp_sy(9), grp_n(9)
712 integer, allocatable :: grp_ray(:,:)
713 double precision :: iib(ixG^T)
714
715 ! The rays are grouped by the signs of (mu_h, mu_v), since the rays of one group share an
716 ! upwind corner and therefore one schedule. Counting the zero sign there are nine sign pairs,
717 ! so no more groups can occur
718 allocate(grp_ray(9, max(rt_sc_nray,1)))
719 ng = 0
720 do k = 1, rt_sc_nray
721 sx = ray_sign(mu_h(k)); sy = ray_sign(mu_v(k))
722 gidx = 0
723 do g = 1, ng
724 if(grp_sx(g)==sx .and. grp_sy(g)==sy) then; gidx = g; exit; end if
725 end do
726 if(gidx==0) then
727 ng = ng + 1
728 grp_sx(ng)=sx; grp_sy(ng)=sy; grp_n(ng)=0; gidx=ng
729 end if
730 grp_n(gidx) = grp_n(gidx) + 1
731 grp_ray(gidx, grp_n(gidx)) = k
732 end do
733 call rt_wave_begin(nisw)
734 do ib = 1, rt_sc_nband
735 call rt_wave_band_open(ib)
736 ! In group gg a block sits at stage p+q, and at clock tt_clk it sweeps ray
737 ! jj = tt_clk-(p+q)+2 of the group into slot i_isw+(gg-gbase). One getbc per clock
738 ! advances every front of the batch by one stage. The accumulators were zeroed when the band
739 ! opened, so they sum over the batches
740 do gbase = 1, ng, nisw
741 gend = min(gbase+nisw-1, ng)
742 nslot = gend-gbase+1
743 ! The slots of the batch start from zero intensity
744 do iblk = 1, wv_nblk
745 ps_rt(wv_ig(iblk))%w(ixmlo1:ixmhi1,ixmlo2:ixmhi2,i_isw:i_isw+nisw-1) = 0.0d0
746 end do
747 ! The last ray of a group reaches the last diagonal at clock grp_n + wv_dmax - 2, and the
748 ! batch runs until its longest group has finished
749 tmax = wv_dmax - 2
750 do gg = gbase, gend; tmax = max(tmax, grp_n(gg) + wv_dmax - 2); end do
751 ! The datatype is built once per batch, for its number of slots
752 call create_bc_mpi_datatype(i_isw, nslot)
753 do tt_clk = 1, tmax
754 ! One getbc moves every front, and the band-end and edge inflows are then written into
755 ! each slot
756 call rt_exchange(nslot)
757 do gg = gbase, gend
758 call rt_inflow_ghosts(i_isw+(gg-gbase))
759 end do
760 do gg = gbase, gend
761 islot = i_isw+(gg-gbase)
762 do iblk = 1, wv_nblk
763 ! jj = t - (p + q) + 2, the ray of the group that reaches this block at clock t. A
764 ! block sweeps at this clock only when that is a ray of the group
765 jj = tt_clk - (wave_stage(wv_bi(iblk),rg_nbh,grp_sx(gg)) &
766 + wave_stage(wv_bj(iblk),wv_nbxv,grp_sy(gg))) + 2
767 if(jj < 1 .or. jj > grp_n(gg)) cycle
768 k = grp_ray(gg, jj)
769 igrid = wv_ig(iblk)
770 ! The copy keeps the upwind ghosts from getbc, and its interior is cleared for ray k
771 iib = ps_rt(igrid)%w(:,:,islot)
772 iib(ixmlo1:ixmhi1,ixmlo2:ixmhi2) = 0.0d0
773 if(wv_seam) call rt_seam_read(iib, igrid, wv_bi(iblk), grp_sx(gg), k)
774 call rt_sweep_one(iblk, k, iib)
775 ! The interior goes back into the slot, the seam edge is kept, and the weighted
776 ! intensity adds to J, Fy and Fx straight away
777 ps_rt(igrid)%w(ixmlo1:ixmhi1,ixmlo2:ixmhi2,islot) = iib(ixmlo1:ixmhi1,ixmlo2:ixmhi2)
778 if(wv_seam) call rt_seam_write(iib, igrid, wv_bi(iblk), grp_sx(gg), k)
779 call rt_add_moments(iblk, k, iib)
780 end do
781 end do
782 end do
783 end do
784 call rt_wave_band_close(nisw)
785 end do
786 call rt_wave_end()
787 end subroutine rt_sc_solve_wave_pipe
